+#[cfg(all(test, windows))]
+#[test]
+fn pending_overwrite_goes_through_the_queue_predicate() {
+    assert!(crate::host::late());
+    let mut readings = vec![];
+    for (memory, stores) in [
+        (1u64, &[][..]),
+        (0, &[0u64, 1][..]),
+        (0, &[1u64, 0][..]),
+        (1, &[1u64, 0][..]),
+    ] {
+        let address = std::ptr::from_ref(&memory) as usize;
+        let w = ValueWait {
+            address: address as u64,
+            comparison: 3,
+            reference: 1,
+            wide: true,
+        };
+        crate::host::audit_pending_words_at(address, stores);
+        readings.push((
+            crate::host::late_met(&w),
+            crate::queue::audit_queue_value_met(&w),
+        ));
+        crate::host::audit_pending_words_at(address, &[]);
+    }
+    eprintln!("WAIT (prediction,queue) {readings:?}");
+    assert_eq!(
+        readings.iter().map(|v| v.1).collect::<Vec<_>>(),
+        [true, true, false, false]
+    );
+}

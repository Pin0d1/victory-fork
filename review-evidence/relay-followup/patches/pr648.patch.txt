+#[cfg(test)]
+mod followup_readback {
+    use super::*;
+    use windows::Win32::Graphics::Dxgi::{
+        CreateDXGIFactory2, DXGI_ADAPTER_FLAG_SOFTWARE, DXGI_CREATE_FACTORY_FLAGS, IDXGIAdapter1,
+        IDXGIFactory4,
+    };
+
+    #[test]
+    fn changed_pieces_preserve_the_producer_after_a_later_direct_write() {
+        let host = host().expect("D3D12 device");
+        let desc = unsafe {
+            let factory: IDXGIFactory4 = CreateDXGIFactory2(DXGI_CREATE_FACTORY_FLAGS(0)).unwrap();
+            let ad: IDXGIAdapter1 = factory
+                .EnumAdapterByLuid(host.device.GetAdapterLuid())
+                .unwrap();
+            ad.GetDesc1().unwrap()
+        };
+        let soft = desc.Flags & DXGI_ADAPTER_FLAG_SOFTWARE.0 as u32 != 0;
+        assert_eq!(soft, std::env::var_os("VICTORY_D3D12_WARP").is_some());
+        eprintln!(
+            "ADAPTER {:04x}:{:04x} software={soft}",
+            desc.VendorId, desc.DeviceId
+        );
+        const PIECE: usize = 65536;
+        let pages: usize = std::env::var("PROBE_PIECES").unwrap().parse().unwrap();
+        let n = pages * PIECE;
+        let mapped = n.max(1 << 20);
+        let mut backing = vec![0u32; (mapped + PIECE) / 4];
+        let address = (backing.as_mut_ptr() as usize).next_multiple_of(PIECE);
+        let start = (address - backing.as_ptr() as usize) / 4;
+        let words = &mut backing[start..start + mapped / 4];
+        crate::shadow::register(address, address as u64, mapped as u64);
+        let shadow = crate::shadow::find(address as u64, 4).unwrap();
+        assert!(shadow.carded());
+        let write = |list: &HostList, v: u32| {
+            let (upload, map) = host.upload_buffer(n as u64).unwrap();
+            unsafe {
+                std::ptr::write_bytes(map, 0, n);
+                for i in 0..pages {
+                    map.add(i * PIECE).cast::<u32>().write_unaligned(v);
+                }
+                list.list
+                    .CopyBufferRegion(&shadow.buffer, 0, &upload, 0, n as u64);
+                list.list.ResourceBarrier(&[transition(
+                    &shadow.buffer,
+                    D3D12_RESOURCE_STATE_COPY_DEST,
+                    D3D12_RESOURCE_STATE_COMMON,
+                )]);
+            }
+            list.keep(&upload);
+        };
+        let overwrite = |v| {
+            let l = host.new_list().unwrap();
+            write(&l, v);
+            l.close();
+            unsafe {
+                let f: ID3D12Fence = host.device.CreateFence(0, D3D12_FENCE_FLAG_NONE).unwrap();
+                host.queue
+                    .ExecuteCommandLists(&[Some(l.list.cast().unwrap())]);
+                host.queue.Signal(&f, 1).unwrap();
+                f.SetEventOnCompletion(1, Default::default()).unwrap();
+            }
+        };
+        let flags = |p: &CardPlan| {
+            let w = CARD_WORK.lock().unwrap();
+            let base = w.as_ref().unwrap().differs.1 as *const u32;
+            let r = &p.ranges[0];
+            assert!(r.compared);
+            assert_eq!(r.span.pieces as usize, pages);
+            unsafe { std::slice::from_raw_parts(base.add(r.at as usize), pages).to_vec() }
+        };
+        let producer = host.new_list().unwrap();
+        shadow.refresh(&producer, address as u64, n as u64);
+        shadow.gpu_writes(&producer, address as u64, n as u64);
+        write(&producer, 17);
+        producer.close();
+        let (lock, mut first) = host.submit(&[&producer], true).unwrap();
+        let fence1 = *lock;
+        drop(lock);
+        let tracked = std::env::var_os("PROBE_TRACKED_LATER").is_some();
+        let second = if tracked {
+            let list = host.new_list().unwrap();
+            shadow.gpu_writes(&list, address as u64, n as u64);
+            write(&list, 29);
+            list.close();
+            let (lock, sent) = host.submit(&[&list], true).unwrap();
+            let done = *lock;
+            drop(lock);
+            unsafe {
+                host.fence
+                    .SetEventOnCompletion(done, Default::default())
+                    .unwrap();
+            }
+            Some((list, sent, done))
+        } else {
+            overwrite(29);
+            None
+        };
+        let p = first.card.take().unwrap();
+        let fs = flags(&p);
+        assert_eq!(fs.iter().filter(|&&v| v != 0).count(), pages);
+        let staged = fs.iter().filter(|&&v| v & 0x8000_0000 != 0).count();
+        let overflow = fs.iter().filter(|&&v| v & 0x4000_0000 != 0).count();
+        let mut fetch = None;
+        host.relay_card_back(&[&p], &mut fetch);
+        let count =
+            |words: &[u32], v: u32| (0..pages).filter(|i| words[i * PIECE / 4] == v).count();
+        let first17 = count(words, 17);
+        let first29 = count(words, 29);
+        if let Some((_list, mut second, fence2)) = second {
+            host.card_done(p);
+            for (s, rs) in first.back {
+                s.copy_back_done(&rs);
+            }
+            let p = second.card.take().unwrap();
+            assert_eq!(flags(&p).iter().filter(|&&v| v != 0).count(), pages);
+            host.relay_card_back(&[&p], &mut fetch);
+            let second29 = count(words, 29);
+            host.card_done(p);
+            for (s, rs) in second.back {
+                s.copy_back_done(&rs);
+            }
+            crate::shadow::unregister(address, address as u64);
+            let error = host.log_messages();
+            eprintln!(
+                "TRACKED pages={pages} staged={staged} overflow={overflow} first17={first17} first29={first29} second29={second29} debug_error={error:?}"
+            );
+            assert_eq!(second29, pages);
+            assert_eq!(
+                first17, pages,
+                "the first landing must see its producer after the second tracked flight"
+            );
+            return;
+        }
+        overwrite(17);
+        host.card_done(p);
+        for (s, rs) in first.back {
+            s.copy_back_done(&rs);
+        }
+        let idle = host.new_list().unwrap();
+        shadow.gpu_writes(&idle, address as u64, n as u64);
+        idle.close();
+        let (lock, mut again) = host.submit(&[&idle], true).unwrap();
+        let fence2 = *lock;
+        drop(lock);
+        unsafe {
+            host.fence
+                .SetEventOnCompletion(fence2, Default::default())
+                .unwrap();
+        }
+        let p = again.card.take().unwrap();
+        let changed = flags(&p).iter().filter(|&&v| v != 0).count();
+        host.relay_card_back(&[&p], &mut fetch);
+        let second17 = count(words, 17);
+        let second29 = count(words, 29);
+        host.card_done(p);
+        for (s, rs) in again.back {
+            s.copy_back_done(&rs);
+        }
+        crate::shadow::unregister(address, address as u64);

+        assert_eq!(
